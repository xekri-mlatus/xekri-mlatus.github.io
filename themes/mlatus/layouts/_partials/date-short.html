{{- /* "3 Nov, 04:45" on tiles; the year is added when it isn't the year of the build ("10 Jun 2017, 12:00"). Dates are shown as written (+03:00). A note without a time (midnight) shows just the day. */ -}} {{- $d := .Date -}} {{- $s := $d.Format "2 Jan" -}} {{- if ne $d.Year now.Year -}}{{- $s = $d.Format "2 Jan 2006" -}}{{- end -}} {{- if ne ($d.Format "15:04:05") "00:00:00" -}}{{- $s = printf "%s, %s" $s ($d.Format "15:04") -}}{{- end -}} {{- return $s -}}
