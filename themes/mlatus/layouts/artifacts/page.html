{{- /* An artifact (content/artifacts//): a page made elsewhere, served as it is, in its own design. Its HTML file sits next to index.md. The site adds only a description for link previews, the visitor's light / dark choice, the site's icon (unless the page has its own), and a thin bar at the top leading back to the site. No baseof: the artifact is a whole page of its own. */ -}} {{- $page := .Resources.GetMatch "*.html" -}} {{- if not $page }}{{ errorf "%s: the artifact's HTML file is missing" .File.Path }}{{ end -}} {{- $icons := "" -}} {{- if not (findRE `]+rel=["']?(shortcut )?icon` $page.RawContent 1) }}{{ $icons = partial "icons.html" . }}{{ end -}} {{- $head := printf ` %s ` .Description .Permalink site.Title .Permalink .Title .Description $icons -}} {{- $bar := printf `${0}
← %s/Artifacts
` site.Title ("" | relURL) site.Title (.Parent.RelPermalink) -}} {{- $html := replace $page.RawContent "" $head 1 -}} {{- replaceRE `]*>` $bar $html 1 | safeHTML -}}
